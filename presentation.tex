\begin{frame}{\textbf{Introduction: Preeclampsia Detection via RepNet}}
    
    % Section 1: The Clinical Problem
    \textbf{Clinical Motivation}
    \begin{itemize}
        \item \textbf{Preeclampsia:} A severe hypertensive pregnancy disorder that can lead to maternal and fetal mortality.
        \item \textbf{Complications:} Liver damage, low platelet count, and preterm birth.
        \item \textbf{Risk of Eclampsia:} Can rapidly progress to affect brain function, putting the mother at risk of seizures or coma.
    \end{itemize}
    
    \vspace{0.3cm} % Clean spacing between sections
    
    % Section 2: The Proposed Machine Learning Solution
    \textbf{Our Approach: RepNet}
    \begin{itemize}
        \item A pair of machine learning models combining 12-lead ECGs and clinical data to estimate preeclampsia risk:
        \begin{itemize}
            \item \textbf{RepNet-RF:} Decision-tree modeling trained on engineered features and clinical data.
            \item \textbf{RepNet-SE:} A depth-wise separable CNN with squeeze-excitation layers and cross-lead attention mechanisms trained on raw ECG waveforms.
        \end{itemize}
        \item \textbf{Dataset:} Evaluate on 2,191 clinical 12-lead ECGs (337 PreE, 15.4\%) across 1,383 patients at UC Davis Health.
    \end{itemize}

\end{frame}

      \usebeamerfont{footline}\color{coolgray}
      \textbf{Cross-Validation Strategy:}
      \begin{itemize}
        \item\textbf{StratifiedGroupKFold} with 5 folds grouped by the Patient's MRN
        \item\textbf{RandomForest} with \textbf{RandomUnderSampler} inside the folds
      \end{itemize}

      \vspace{0.2em}
      \textbf{Feature Selection:}
      \begin{itemize}
        \item \textbf{Pearson Correlation:} Drops one feature out of colinear pairs sharing a correlation value $> 0.85$.

        \item \textbf{RFECV:} Used roc\_auc scoring to recursively remove features

    \begin{column}{0.48\textwidth}

        \textbf{Lead Reduction \& Wearables:}
        \begin{itemize}

        \end{itemize}
      \end{beamercolorbox}

        \textbf{Transfer Learning:}
        \begin{itemize}
          \item Pretraining on large public ECG corpora (PTB-XL, MIMIC-IV-ECG) or leveraging ECG foundation models before fine-tuning on the UCDH cohort.
        \end{itemize}
      \end{beamercolorbox}
