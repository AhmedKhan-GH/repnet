\section{Introduction}
% =========================================================

\begin{frame}{Motivation}
  \begin{itemize}
    \item Preeclampsia: hypertensive disorder of pregnancy, 2--8\% of complications.
    \item Diagnosis currently relies on $\geq$20-week BP and end-organ markers.
    \item \textbf{Goal:} predict risk weeks earlier from 12-lead ECG morphology.
  \end{itemize}
\end{frame}

\begin{frame}{Goal}
  \begin{block}{What we are building}
    An \emph{explainable} preeclampsia risk pipeline pairing engineered ECG
    features with learned representations, where every prediction can be
    traced back to a clinically meaningful quantity.
  \end{block}
\end{frame}
