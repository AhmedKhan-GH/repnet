% ---------- Introduction Slide ----------
\begin{frame}{Introduction}
  \begin{columns}[T]

    \begin{column}{0.52\textwidth}
      \usebeamerfont{footline}\color{coolgray}
      {\small\Raleway\bfseries\color{aggieblue} What is Preeclampsia?}\\[0.3em]
      \begin{itemize}
        \item Pregnancy complication with dangerously high blood pressure and organ stress
        \item Typically develops after 20 weeks of gestation
        \item Can damage kidneys, liver, and brain --- may progress to eclampsia (seizures and coma)
        \item Currently diagnosed by symptoms that appear \textbf{late} (elevated BP, proteinuria)
      \end{itemize}

      \vspace{0.6em}
      {\small\Raleway\bfseries\color{aggieblue} Why ECG?}\\[0.3em]
      \begin{itemize}
        \item PE increases cardiac afterload --- leaves traces in the ECG waveform
        \item 12-lead ECG is routine, cheap, and already collected in prenatal care
        \item Goal: detect PE risk from the ECG \textbf{before} symptoms appear
      \end{itemize}
    \end{column}

    \begin{column}{0.44\textwidth}
      {\small\Raleway\bfseries\color{aggieblue} Our Approach}\\[0.3em]
      \usebeamerfont{footline}\color{coolgray}
      Two parallel modeling strategies on 2{,}178 ECG recordings (335 PE+, 1{,}843 PE$-$):\\[0.3em]

      \begin{beamercolorbox}[wd=\textwidth,rounded=true,leftskip=0.5em,rightskip=0.5em,sep=0.4em]{boxgray}
        \scriptsize\Lato\color{coolgray}
        \textbf{1.\ Decision-Tree Modeling}\\
        Handcrafted features (NeuroKit + MUSE) $\rightarrow$ feature selection $\rightarrow$ Random Forest \& ensemble classifiers
      \end{beamercolorbox}

      \vspace{0.4em}
      \begin{beamercolorbox}[wd=\textwidth,rounded=true,leftskip=0.5em,rightskip=0.5em,sep=0.4em]{boxgray}
        \scriptsize\Lato\color{coolgray}
        \textbf{2.\ Neural Network Modeling}\\
        Raw 12-lead waveform $\rightarrow$ PerLeadCNN (per-lead shared-weight CNN, 29K params, selected from 250 autonomous experiments)
      \end{beamercolorbox}

      \vspace{0.4em}
      \begin{beamercolorbox}[wd=\textwidth,rounded=true,leftskip=0.5em,rightskip=0.5em,sep=0.3em]{boxgray}
        \scriptsize\Lato\color{coolgray}
        \textbf{Evaluation:} Patient-grouped cross-validation, StratifiedGroupKFold, 80/20 dev-test split
      \end{beamercolorbox}
    \end{column}

  \end{columns}
\end{frame}

\begin{frame}{Decision-Tree Modeling --- Feature Importance}
  \begin{columns}[T]
    
    % Left Column: Methodological Framework details
    \begin{column}{0.42\textwidth}
      \usebeamerfont{footline}\color{coolgray}
      \textbf{Cross-Validation Strategy:}
      \begin{itemize}
        \item\textbf{StratifiedGroupKFold} with 5 folds grouped by the Patient's MRN
        \item\textbf{RandomForest} with \textbf{RandomUnderSampler} inside the folds
      \end{itemize}

      \vspace{0.2em}
      \textbf{Feature Selection:}
      \begin{itemize}
        \item \textbf{Pearson Correlation:} Drops one feature out of colinear pairs sharing a correlation value $> 0.85$.
        \item \textbf{ANOVA (Analysis of Variance):} Keeps features with $p < 0.05$.
        \item \textbf{RFECV:} Used roc\_auc scoring to recursively remove features
      \end{itemize}
      
      \vspace{0.2em}
      \begin{beamercolorbox}[wd=\textwidth,rounded=true,leftskip=0.5em,rightskip=0.5em,sep=0.3em]{boxgray}

  \begin{columns}[T]

    \begin{column}{0.48\textwidth}
      \centering
      {\small\Raleway\bfseries\color{aggieblue} Signal Preprocessing}\\[0.3em]
      \usebeamerfont{footline}\color{coolgray}
      Three steps applied per-lead in order:
      \begin{enumerate}\setlength{\itemsep}{0pt}
        \item \textbf{Baseline Wander Removal:} 4th-order Butterworth high-pass at 0.5~Hz
        \item \textbf{Powerline Interference:} 60~Hz IIR notch filter ($Q=30$)
        \item \textbf{Z-Score Normalization:} Per-lead $\frac{x - \mu_\ell}{\sigma_\ell + \epsilon}$
      \end{enumerate}
    \end{column}

    \begin{column}{0.48\textwidth}
      {\small\Raleway\bfseries\color{aggieblue} Training Configuration}\\[0.3em]
      \usebeamerfont{footline}\color{coolgray}

        \item \textbf{AdamW} + cosine annealing (lr $= 1.2\!\times\!10^{-3}$)
        \item Dropout 0.15, weight decay $5\!\times\!10^{-3}$
        \item Focal loss ($\gamma = 1.0$) + label smoothing 0.05
        \item Early stop on val AUROC, patience 20, 80 epochs
        \item \textbf{30-split} patient-grouped validation (80/20)
      \end{itemize}

      An autonomous AI agent (Karpathy's \textit{autoresearch} framework) iteratively edited the model code, trained, and evaluated --- keeping improvements and discarding regressions:
      \begin{enumerate}
        \item Modify \texttt{train.py} with a new architectural idea
        \item Train for up to 80 epochs (10-min wall-clock budget)
        \item Evaluate on held-out validation split
        \item If AUROC improves $\to$ keep; otherwise $\to$ discard
        \item Repeat (250 experiments total)

\begin{frame}{Conclusion}
  {\small\Raleway\bfseries\color{aggieblue} Key Findings}\par\vspace{0.3em}
  \usebeamerfont{footline}\color{coolgray}
  \begin{itemize}\setlength{\itemsep}{2pt}\setlength{\parskip}{0pt}
    \item 12-lead ECGs contain a detectable \textbf{preeclampsia signal} --- both feature-engineered and neural approaches exceed chance on 30-split patient-grouped validation
    \item Decision-tree ensemble (\textbf{Unweighted Voting+}): \textbf{AUROC $75.72 \pm 3.94$}, balanced accuracy $68.50 \pm 3.22$ --- stable baseline on 175 handcrafted features
    \item PerLeadCNN: \textbf{AUROC $0.702 \pm 0.043$}, \textbf{80.7\% sensitivity}, \textbf{NPV 92.8\%} at screening threshold --- only \textbf{29{,}490 parameters} on raw waveforms
    \item After 250 autonomous experiments, \textbf{simplicity dominated}: SE, attention, skip connections, and frequency-domain branches all failed to improve over a basic per-lead CNN
    \item Grad-CAM highlights \textbf{lead aVL} and \textbf{P-wave / QT-interval} regions, consistent with PE-associated ECG changes in literature
  \end{itemize}

  \vspace{0.3em}
  \begin{beamercolorbox}[wd=\textwidth,rounded=true,leftskip=0.5em,rightskip=0.5em,sep=0.3em]{boxgray}
    \scriptsize\Lato\color{coolgray}
    \textbf{Takeaway:} ECG-based PE screening is \textbf{feasible}. The $\sim$0.70 AUROC ceiling is data-limited ($n{=}335$ PE+), not model-limited --- larger, multi-site cohorts are the path to clinical utility.
  \end{beamercolorbox}
\end{frame}
