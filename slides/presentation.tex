\begin{frame}{Neural Network Modeling --- Training}

  \begin{large}
  \begin{itemize}
    \setlength{\itemsep}{1.0em}

    \item \textbf{80/20 dev-test split} with StratifiedGroupKFold --- preserves clinical prevalence, no patient leakage

    \item \textbf{AdamW} with cosine annealing --- early epochs explore, late epochs refine; early stop on val AUROC (patience 20)

    \item \textbf{7 on-the-fly augmentations} --- noise, scaling, time shift, lead dropout, cutout, baseline wander, speed perturbation

    \item \textbf{Per-lead preprocessing:} 0.5~Hz high-pass, 60~Hz notch filter, z-score normalization

  \end{itemize}
  \end{large}
